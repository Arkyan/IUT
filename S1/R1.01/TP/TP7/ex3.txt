fonction saisie() retourne listnom[100]
début 
    listnom : tableau[100] 
    nom : chaine
    encore : chaine
    encore <- "o"
    listnom <- []

    tant que encore = "o" faire
        saisir nom
        tantque nom = "" faire
            saisir nom
        fin tantque

        listnom.ajouter(nom)
        Afficher 'Voulez-vous saisir un autre nom ? (o = oui/n = non) : \n Saisir o ou n : '
        saisir encore
    fin tantque
    retourne listnom
fin

fonction voyelle(chaine : chaine) retourne entier
début
    voyelles : tableau[100] de chaine
    nb_voyelles : entier
    lettre : chaine

    voyelles <- ['a', 'e', 'i', 'o', 'u', 'y', 'A', 'E', 'I', 'O', 'U', 'Y']
    nb_voyelles <- 0
    pour lettre dans chaine faire
        si lettre dans voyelles alors
            nb_voyelles <- nb_voyelles + 1
        fin si
    fin pour
    retourne nb_voyelles
fin

fonction palindrome(chaine : chaine) retourne booleen
debut
    chaine <- chaine.remplacer(" ", "")
    retourne chaine == chaine[::-1]ù
fin

---------------------------------------------------------------------------

fonction afficherPalindromes(listnom : listnom[100]) -> None
début
    pour nom dans listnom faire
        si palindrome(nom) alors
            Afficher 'Le nom -', nom ,'- est un palindrome'
        fin si
    fin pour
fin
